\begin{frame}{REVISÃO DE LITERATURA}
  \framesubtitle{Fundamentação Matemática: Conceitos Iniciais}
  
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{block}{Definições Básicas}
        \small
        \begin{itemize}
            \item \textbf{Grafo $\mathcal{G} = (\mathcal{V}, \mathcal{A})$:} Estrutura matemática definida por um conjunto não vazio de \textbf{vértices} ($\mathcal{V}$) e um conjunto de \textbf{arestas} ($\mathcal{A}$), que estabelecem as relações de adjacência entre os pares de vértices.
            \vspace{0.3cm}
            \item \textbf{Grau de um Vértice $\delta(v)$:} Definido rigorosamente como o número de arestas incidentes no vértice $v$.
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \small \textbf{Exemplo adjacente:} Observa-se que o vértice $A$ possui três arestas incidentes, portanto, seu grau é 3 ($\delta(A)=3$).
    \end{column}
    
    \begin{column}{0.45\textwidth}
      \begin{center}
        \begin{tikzpicture}[scale=1.5, auto, swap, >=stealth]

% Slide 8: Revisão de Literatura | Fundamentação Matemática (Proposta I)
\begin{frame}{REVISÃO DE LITERATURA}
  \framesubtitle{Fundamentação Matemática: Proposta I ("A Vila dos Grafos")}
  \begin{tcolorbox}[colback=white, colframe=uescgreen, title=Lema 3.1: Lema do Aperto de Mão]
    Em qualquer grafo $\mathcal{G}=(\mathcal{V},\mathcal{A})$, a soma dos graus de todos os vértices é igual ao dobro do número de arestas:
    $$\sum_{v\in\mathcal{V}}\delta(v)=2|\mathcal{A}|$$
  \end{tcolorbox}
  
  \vspace{0.1cm}
  
  \begin{block}{Demonstração}
    \begin{enumerate}
        \item Considere a construção de um grafo. Cada aresta $\{u,v\}$ conecta exatamente dois vértices distintos.
        \item Ao traçarmos uma aresta, ela contribui com exatamente uma unidade ($+1$) para o grau do vértice $u$ e outra unidade ($+1$) para o vértice $v$.
        \item Portanto, ao percorrermos todas as arestas para calcular a soma total dos graus, \textbf{cada aresta será contabilizada exatamente duas vezes}.
        \item Logo, a soma dos graus é obrigatoriamente o dobro da quantidade total de arestas. \qedsymbol
    \end{enumerate}
  \end{block}
\end{frame}

% Slide 9: Revisão de Literatura | Fundamentação Matemática (Proposta II)
\begin{frame}{REVISÃO DE LITERATURA}
  \framesubtitle{Fundamentação Matemática: Proposta II ("Redes Sociais")}
  \begin{tcolorbox}[colback=white, colframe=uescgreen, title=Dígrafos e Conservação de Grau]
    Em um Grafo Dirigido (Dígrafo), as arestas possuem orientação (arcos). A soma dos graus de entrada ($\delta_{in}$) é igual à soma dos graus de saída ($\delta_{out}$), sendo ambas iguais ao total de arcos $|\mathcal{A}|$.
  \end{tcolorbox}
  
  \vspace{0.1cm}
  
  \begin{block}{Demonstração Lógica}
    \begin{enumerate}
        \item Em redes assimétricas (como seguidores no Instagram), um arco direcionado parte de um vértice de origem e aponta para um vértice de destino.
        \item A existência de um único arco gera simultaneamente \textbf{exatamente um} grau de saída na origem e \textbf{exatamente um} grau de entrada no destino.
        \item Sendo assim, cada arco do conjunto $|\mathcal{A}|$ contribui com $1$ para o somatório total de saídas e com $1$ para o somatório total de entradas.
        \item Conclui-se que $\sum \delta_{in} = \sum \delta_{out} = |\mathcal{A}|$. \qedsymbol
    \end{enumerate}
  \end{block}
\end{frame}

% Slide 10: Revisão de Literatura | Fundamentação Matemática (Proposta III)
\begin{frame}{REVISÃO DE LITERATURA}
  \framesubtitle{Fundamentação Matemática: Proposta III ("Caixeiro Viajante")}
  \begin{tcolorbox}[colback=white, colframe=uescgreen, title=Teorema 3.1: Arestas no Grafo Completo]
    O grafo completo $K_n$ possui exatamente $\frac{n(n-1)}{2}$ arestas. Cada vértice possui grau $n-1$.
  \end{tcolorbox}
  
  \vspace{0.1cm}
  
  \begin{block}{Demonstração}
    \begin{enumerate}
        \item No grafo completo $K_n$, todos os vértices estão conectados entre si. Logo, cada um dos $n$ vértices liga-se aos outros $n-1$ vértices disponíveis.
        \item O grau de qualquer vértice é, portanto, $n-1$.
        \item A soma dos graus de todos os $n$ vértices será $n \times (n-1)$.
        \item Aplicando o Lema do Aperto de Mão, sabemos que a soma dos graus é igual a $2|\mathcal{A}|$.
        \item Igualando os termos: $2|\mathcal{A}| = n(n-1) \implies |\mathcal{A}| = \frac{n(n-1)}{2}$. \qedsymbol
    \end{enumerate}
  \end{block}
\end{frame}

% Slide 11: Metodologia da Pesquisa
\begin{frame}{METODOLOGIA DA PESQUISA}
  \textbf{Design do Estudo:} Pesquisa qualitativa, exploratória e bibliográfica, cujo produto final é a estruturação de um referencial pedagógico viável.
  
  \vspace{0.4cm}
  
  \begin{block}{Os Três Pilares: Propostas Didáticas Progressivas}
    \begin{enumerate}
      \item \textbf{A Vila dos Grafos (Proposta I):} Atividade de mapeamento viário para construção de vértices/arestas e constatação dedutiva do Lema do Aperto de Mão.
      \item \textbf{Redes Sociais (Proposta II):} Aplicação de Dígrafos para diferenciar amizades simétricas (Facebook) de conexões assimétricas (Instagram).
      \item \textbf{Caixeiro Viajante (Proposta III):} Cálculo de rotas logísticas mais curtas com arestas ponderadas e reflexão sobre a complexidade computacional.
    \end{enumerate}
  \end{block}
\end{frame}
